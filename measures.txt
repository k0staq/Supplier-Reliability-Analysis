-- On-time Delivery %
OTD % =
VAR TotalOrders =
    COUNTROWS ( FactOrders )
VAR OnTimeOrders =
    CALCULATE (
        COUNTROWS ( FactOrders ),
        FactOrders[DeliveryStatus] = "On-time"
    )
RETURN
DIVIDE ( OnTimeOrders, TotalOrders )

-- Delay %
Delay % =
VAR TotalOrders =
    COUNTROWS ( FactOrders )
VAR DelayedOrders =
    CALCULATE (
        COUNTROWS ( FactOrders ),
        FactOrders[DeliveryStatus] = "Delayed"
    )
RETURN
DIVIDE ( DelayedOrders, TotalOrders )

-- Average Lead Time (days)
Avg Lead Time (days) =
AVERAGEX ( FactOrders, FactOrders[LeadTimeDays] )

-- Delay Count
Delay Count =
CALCULATE (
    COUNTROWS ( FactOrders ),
    FactOrders[DeliveryStatus] = "Delayed"
)

-- Total Delay Cost (€)
Total Delay Cost (€) =
SUM ( FactOrders[DelayCostEUR] )

-- Avg Cost per Delayed Order (€)
Avg Cost per Delayed Order (€) =
DIVIDE ( [Total Delay Cost (€)], [Delay Count] )

-- Worst Supplier (by delay cost)
Worst Supplier =
VAR t =
    TOPN (
        1,
        SUMMARIZE (
            FactOrders,
            DimSuppliers[SupplierName],
            "DelayCost", [Total Delay Cost (€)]
        ),
        [DelayCost], DESC
    )
RETURN
MAXX ( t, DimSuppliers[SupplierName] )

-- Top Cost Cause (Reason)
Top Cost Cause (Reason) =
VAR t =
    TOPN (
        1,
        SUMMARIZE (
            FactOrders,
            DimDelayReasons[ReasonName],
            "DelayCost", [Total Delay Cost (€)]
        ),
        [DelayCost], DESC
    )
RETURN
MAXX ( t, DimDelayReasons[ReasonName] )

-- Top Frequency Cause (Reason)
Top Frequency Cause (Reason) =
VAR t =
    TOPN (
        1,
        SUMMARIZE (
            FactOrders,
            DimDelayReasons[ReasonName],
            "DelayCount", [Delay Count]
        ),
        [DelayCount], DESC
    )
RETURN
MAXX ( t, DimDelayReasons[ReasonName] )

-- Supplier Risk Tier (column example on DimSuppliers)
Supplier Risk Tier =
VAR CostVal = [Total Delay Cost (€)]
VAR DelayCnt = [Delay Count]
RETURN
SWITCH (
    TRUE (),
    CostVal > 20000 && DelayCnt > 10, "1 - 🔥 Critical",
    CostVal > 20000 && DelayCnt <= 10, "2 - ⚠ High Cost",
    CostVal <= 20000 && DelayCnt > 10, "3 - ⚠ High Frequency",
    "4 - 🟢 Low Risk"
)
